main.js 
// Corbit Mono — clean and readable at any size
const fibonacci = (n) => {
  if (n <= 1) return n;
  return fibonacci(n - 1) + fibonacci(n - 2);
};

const results = [0, 1, 2, 3, 4, 5, 6, 7, 8]
  .map(fibonacci);

console.log(`Sequence: ${results.join(', ')}`);

interface Response<T> {
  status: number;
  data: T;
  error?: string;
}

async function fetchUser(id: string): Promise<Response<User>> {
  const res = await fetch(`/api/users/${id}`);
  if (!res.ok) throw new Error('Not found');
  return { status: 200, data: await res.json() };
}

Why Corbit Mono?

◈

Stroke-Based Design

Every glyph built from geometric strokes — precise, legible, and unmistakably technical.

◉

True Monospace

All characters share a fixed 600-unit advance width for perfect code alignment.

◎

Open Source

Released under SIL Open Font License 1.1. Free to use, modify, and distribute.

◍

Developer Symbols

Includes all essential programming punctuation and special characters.

◌

Built with JS

100% generated from SVG path data using opentype.js — fully reproducible build.

◆

No Dependencies

This demo page is self-contained — no frameworks, no CDN, no trackers.
